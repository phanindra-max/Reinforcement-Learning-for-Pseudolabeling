% Section 1
\section{Introduction}
\begin{frame}{Introduction}
  \begin{itemize}
    \item \textbf{Problem:} Semi-supervised learning is valuable when labeled data is limited, but manually labeling large datasets is time-consuming and costly.
    \item \textbf{Challenge:} Traditional pseudo-labeling heuristics may propagate errors from mislabeled data, leading to reduced model performance.
    \item \textbf{Idea:} Apply Reinforcement Learning (RL) to develop an adaptive pseudo-labeling strategy that selects informative examples, balances risk and reward, and improves model accuracy while reducing labeling errors.
  \end{itemize}
\end{frame}

% Background Section
\section{Background and Literature Review}
\begin{frame}{Background and Literature Review}
    \begin{itemize}
        \item Reinforcement Learning offers a dynamic and adaptive approach to pseudo-labeling, potentially overcoming the limitations of static, heuristic-based methods.
        \item With limited prior experience in RL frameworks, our team initiated a foundational learning phase by implementing starter examples from the Gymnasium library.
        \item To build a strong research context, we subsequently implemented and analyzed five key publications relevant to RL-based pseudo-labeling before developing our own model.
    \end{itemize}
\end{frame}

% Section 3
\section{Reinforcement Learning Framework}
\begin{frame}{MDP Formulation}
  \begin{block}{State}
    Image features, model predictions, entropy (uncertainty).
  \end{block}
  \begin{block}{Action}
    Assign pseudo-label or skip.
  \end{block}
  \begin{block}{Reward}
    Positive for correct pseudo-labels, penalty for wrong ones.
  \end{block}
\end{frame}

% Section 4
\section{Training Flow}
\begin{frame}{Workflow}
  \begin{enumerate}
    \item Load labeled and unlabeled MNIST data.
    \item Extract features with CNN.
    \item RL agent selects pseudo-label actions.
    \item Retrain classifier with newly labeled data.
  \end{enumerate}
\end{frame}

\section{Results and Insights}
\begin{frame}{Observations}
  \begin{itemize}
    \item RL policy learns to prefer high-confidence samples.
    \item Entropy and softmax probabilities guide state representation.

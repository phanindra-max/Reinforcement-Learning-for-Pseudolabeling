\section{Reinforcement Learning Framework}

\begin{frame}{Reinforcement Learning Introduction}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      \begin{itemize}
        \item \textbf{Reinforcement Learning:} Agent learns through interaction and feedback from the environment. \\~\\
        \item \textbf{Sequential Decision Making:} Each action affects future states and rewards. \\~\\
        \item \textbf{Markov Decision Process (MDP):} Framework for modelling decision making.
      \end{itemize}
    \end{column}

    \begin{column}{0.4\textwidth}
      \centering
      \includegraphics[width=\linewidth]{RLPL/images/MDP.png} \\[0.3em]
      {\small\textbf{Markov Decision Process}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{MDP Formulation}
  \begin{block}{State}
    Image features, model predictions, entropy (uncertainty).
    \begin{center}
      $\mathcal{S} = \{ \mathbf{X} ; \ \text{softmax}(\mathbf{y}) ; \ \mathcal{L}  \}$
    \end{center}
  \end{block}
  \begin{block}{Action}
    Assign pseudo-label or skip.
    \begin{center}
      $\mathcal{A} = \{ 0, \ldots, 9, ; \ \text{skip} \}$
    \end{center}
  \end{block}
  \begin{block}{Reward}
    Error Rate improvement of downstream model on validation data.
    \begin{center}
      $\mathcal R_{t} = -(e_{t} - e_{t-1})$
    \end{center}

  \end{block}
\end{frame}

\section{Training Flow}
\begin{frame}{Workflow}
  \begin{enumerate}
    \item \textbf{Load} labeled and unlabeled MNIST data. \\~\\
    \item \textbf{Initiate} RL environment and agent.\\~\\
    \item RL agent selects \textbf{pseudo-label} actions. \\~\\
    \item \textbf For RLGSSL: generate mixup samples, compute consistency reward, update classifier/policy jointly. \\~\\
    \item \textbf{Retrain} downstream model with newly labeled data. \\~\\
    \item \textbf{Reward} is assigned based on downstream model's error rate.
  \end{enumerate}
\end{frame}

\section{PseudoCode}
\begin{frame}{PseudoCode}
  \begin{algorithm}[H]
    \caption{RL agent training loop}
    \label{alg:rl_training_algpseudocode}
    \scriptsize
    \begin{algorithmic}
      \Require state\_dim, action\_dim, labeled\_data, unlabeled\_data, num\_episodes
      \State agent $\leftarrow$ \texttt{RLAgent}(state\_dim, action\_dim)
      \State env $\leftarrow$ \texttt{PseudoLabelEnv}(labeled\_data, unlabeled\_data) \\~\\
      \For{episode = 1 \textbf{to} num\_episodes}
      \State state $\leftarrow$ env.reset()
      \State done $\leftarrow$ false
      \While{not done}
      \State action $\leftarrow$ agent.select\_action(state)
      \State $(\text{next\_state},\ \text{reward},\ \text{terminated},\ \text{info}) \leftarrow \text{env.step(action)}$
      % \State agent.remember(state, action, reward, next\_state, done)
      \State state $\leftarrow$ next\_state
      \State done $\leftarrow$ terminated
      \EndWhile
      \State agent.update()
      \State accuracy $\leftarrow$ env.evaluate()
      \EndFor
      \State env.close()
    \end{algorithmic}
  \end{algorithm}
\end{frame}

\section{Results and Insights}
\begin{frame}{Key Observations}
  \begin{itemize}
    \item RL-based pseudo-labeling improves early-stage learning performance.
    \item CNN benefits more from RL guidance than ResNet due to weaker baseline.
    \item Performance degradation observed after multiple episodes.
    \item \textbf{Key Bottleneck:} Reward design and confirmation bias.
  \end{itemize}

\begin{frame}{What We Learned Since Midterm}
  \begin{itemize}
    \item RL improves pseudo-label selection primarily in early epochs.
    \item Performance degradation occurs due to error reinforcement.
    \item Reward shaping has higher impact than network architecture.
    \item CNN shows stronger gains from RL guidance than ResNet.
    \item RL training is significantly more computationally expensive.
  \end{itemize}

\section{Discussion}
\begin{frame}{Discussion}
  \begin{itemize}
    \item RL dynamically adapts pseudo-label selection beyond static confidence thresholds.
    \item Early performance gains confirm RL’s potential for label efficiency.
    \item Degradation highlights the need for long-term reward stabilization.
    \item ResNet stability limits the impact of RL compared to CNN.
  \end{itemize}
\end{frame}

\section{Conclusion}
\begin{frame}{Conclusion}
  \begin{itemize}
    \item Successfully demonstrated RL-based pseudo-labeling in semi-supervised learning.
    \item RL improves early learning by selecting high-confidence unlabeled samples.
    \item Long-term training stability remains a critical challenge.
    \item Reward design plays a central role in RL-based pseudo-labeling.
    \item The framework shows strong research potential but needs stability optimization.
  \end{itemize}
\end{frame}

%------------------------------

\section{Limitations}
\begin{frame}{Limitations}
  \begin{itemize}
    \item \textbf{Performance Degradation}: Accuracy drops after several episodes due to confirmation bias.
    \item \textbf{Reward Sensitivity}: Small reward changes cause unstable training.
    \item \textbf{Computational Cost}: RL training is 3--4$\times$ slower than standard pseudo-labeling.
    \item \textbf{Noisy Labels}: Early incorrect pseudo-labels propagate errors.
  \end{itemize}
\end{frame}
